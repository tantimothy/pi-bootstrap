# aalib ships aafire (the burning-terminal splash) and is also what bb
# renders through — see bin/install-bb, which builds bb itself since
# nothing packages it for macOS. Note that this bottle has no terminal
# driver at all (only `stdout`), so bin/install-aalib rebuilds it with one;
# this line still earns its place as the quickest way to get an aafire on
# the machine, and as the fallback when that build can't run.
aalib

# hollywood itself has no Homebrew formula (it's a Debian package
# upstream), so run.sh fetches it — see its own comment. These three are
# what its macOS-viable widgets need: figlet for the banner pane, pv for
# the transfer gauges, htop for the process monitor. Every other upstream
# widget wants ccze (unmaintained, no formula) or a Linux-only tool, and
# is deliberately not fetched.
figlet
pv
htop
